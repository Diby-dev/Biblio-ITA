CREATE TABLE utilisateur (
  id_utilisateur INT AUTO_INCREMENT PRIMARY KEY,
  nom_utilisateur VARCHAR(100) NOT NULL,
  prenom_utilisateur VARCHAR(100) NOT NULL,
  type_utilisateur ENUM('Etudiant', 'Enseignant', 'Personnel', 'Autre') NOT NULL,
  classe_utilisateur VARCHAR(24),
  filiere_utilisateur VARCHAR(90),
  contact_utilisateur VARCHAR(100) UNIQUE,
  statut_utilisateur ENUM('inscrit', 'désabonné') NOT NULL
);

CREATE TABLE auteur (
  id_auteur INT AUTO_INCREMENT PRIMARY KEY,
  nom_auteur VARCHAR(100) NOT NULL,
  prenom_auteur VARCHAR(100) NOT NULL,
  nationalite_auteur VARCHAR(100),
  date_naissance_auteur DATE
);

CREATE TABLE fournisseur (
  id_fournisseur INT AUTO_INCREMENT PRIMARY KEY,
  nom_fournisseur VARCHAR(255) NOT NULL,
  contact_fournisseur VARCHAR(100) UNIQUE,
  email_fournisseur VARCHAR(100) UNIQUE,
  adresse_fournisseur VARCHAR(50)
);

CREATE TABLE livre (
  id_livre INT AUTO_INCREMENT PRIMARY KEY,
  titre_livre VARCHAR(255) NOT NULL,
  statut_livre ENUM('disponible','emprunté') NOT NULL,
  id_auteur INT,
  id_fournisseur INT,
  FOREIGN KEY (id_auteur) REFERENCES auteur(id_auteur),
  FOREIGN KEY (id_fournisseur) REFERENCES fournisseur(id_fournisseur)
);

CREATE TABLE emprunt (
  id_emprunt INT AUTO_INCREMENT PRIMARY KEY,
  id_utilisateur INT,
  id_livre INT,
  date_emprunt DATE NOT NULL,
  statut_emprunt ENUM('en cours','retourné','en retard') NOT NULL,
  date_limite_retour_emprunt DATE NOT NULL,
  date_retour_emprunt DATE,
  FOREIGN KEY (id_utilisateur) REFERENCES utilisateur(id_utilisateur),
  FOREIGN KEY (id_livre) REFERENCES livre(id_livre)
);
